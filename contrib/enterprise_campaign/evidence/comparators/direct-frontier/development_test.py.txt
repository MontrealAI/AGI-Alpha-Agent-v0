started = time.perf_counter()
records = []
for case in CASES:
    args = {name: case[name] for name in ['before','after','keys','decimal_columns','aliases']}
    record = dict(case)
    try:
        actual = transform(**args)
        record['actual'] = actual
        record['passed'] = not case.get('expect_error', False) and actual == {
            'output_csv': case.get('expected_output_csv'), 'rollback_csv': case['before']}
        if not record['passed']:
            record['diagnostic'] = 'Unexpected output or expected rejection did not occur'
    except Exception as exc:
        record['exception_type'] = type(exc).__name__
        record['diagnostic'] = str(exc)
        record['passed'] = case.get('expect_error', False) and isinstance(exc, TransformError)
    records.append(record)
report = {'recorded_at_utc': datetime.now(timezone.utc).isoformat(),
          'test_runtime_seconds': time.perf_counter() - started,
          'passed': sum(record['passed'] for record in records), 'total': len(records),
          'cases': records}
ROOT.joinpath('development_results.json').write_text(json.dumps(report, indent=2, ensure_ascii=False))
print(json.dumps({key: report[key] for key in ['passed','total','test_runtime_seconds']}))
for record in records:
    if not record['passed']:
        print(record['name'], record.get('diagnostic'), record.get('actual'))
raise SystemExit(report['passed'] != report['total'])
