@pytest.mark.parametrize('existing',[True,False])
def test_invalid_checksum_preserves_destination(module,monkeypatch,tmp_path,existing):
    monkeypatch.setattr(module,'CHECKSUMS',{'asset.bin':'sha256-'+hashlib.sha256(b'valid').hexdigest()})
    dest=tmp_path/'asset.bin'
    if existing:dest.write_bytes(b'prior valid bytes')
    with pytest.raises(RuntimeError,match='Checksum mismatch'):
        module.download('https://test.invalid/asset',dest)
    if existing:assert dest.read_bytes()==b'prior valid bytes'
    else:assert not dest.exists()

@pytest.mark.parametrize('encoding',['hex','base64','no_checksum'])
def test_success_bytes_preserved(module,monkeypatch,tmp_path,encoding):
    digest=hashlib.sha256(b'review payload').digest()
    ref=digest.hex() if encoding=='hex' else base64.b64encode(digest).decode()
    checksums={} if encoding=='no_checksum' else {'asset.bin':'sha256-'+ref}
    monkeypatch.setattr(module,'CHECKSUMS',checksums)
    dest=tmp_path/'asset.bin';module.download('https://test.invalid/asset',dest)
    assert dest.read_bytes()==b'review payload'
